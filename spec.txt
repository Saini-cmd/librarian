# embedder.py — Full Code

```python
from sentence_transformers import SentenceTransformer
from typing import List
from chunking.chunk_model import CodeChunk
import torch


class Embedder:
    def __init__(self, model_name: str = "BAAI/bge-large-en-v1.5"):

        self.device = "cuda" if torch.cuda.is_available() else "cpu"
        print(f"[Embedder] Using device: {self.device}")

        self.batch_size = 64 if self.device == "cuda" else 16

        self.model = SentenceTransformer(
            model_name,
            device=self.device
        )

        self.embedding_dim = self.model.get_sentence_embedding_dimension()
        print(f"[Embedder] Embedding dimension: {self.embedding_dim}")

    def embed_texts(self, texts: List[str]):
        return self.model.encode(
            texts,
            normalize_embeddings=True,
            show_progress_bar=True,
            batch_size=self.batch_size
        )

    def embed_chunks(self, chunks: List[CodeChunk]):

        texts = [self._prepare_text(chunk) for chunk in chunks]

        embeddings = self.embed_texts(texts)

        return embeddings

    def _prepare_text(self, chunk: CodeChunk):

        return (
            f"Repository: {chunk.repo}\n"
            f"File: {chunk.file_path}\n"
            f"Language: {chunk.language}\n"
            f"Symbol: {chunk.symbol}\n"
            f"Node Type: {chunk.node_type}\n"
            f"Chunk Source: {chunk.chunk_source}\n"
            f"Lines: {chunk.start_line}-{chunk.end_line}\n\n"
            f"{chunk.content}"
        )
```

# embedding_cache.py Code

```python
import pickle
from pathlib import Path


class EmbeddingCache:
    def __init__(self, cache_dir="embedding_cache"):
        self.cache_dir = Path(cache_dir)
        self.cache_dir.mkdir(exist_ok=True)

    def get(self, chunk_id: str):
        path = self.cache_dir / chunk_id

        if path.exists():
            with open(path, "rb") as f:
                return pickle.load(f)

        return None

    def set(self, chunk_id: str, embedding):
        path = self.cache_dir / chunk_id

        with open(path, "wb") as f:
            pickle.dump(embedding, f)
```

# embedding_pipeline.py Code

```python
from typing import List

from chunking.chunk_model import CodeChunk
from .embedder import Embedder
from .embedding_cache import EmbeddingCache


class EmbeddingPipeline:
    def __init__(self):
        self.embedder = Embedder()
        self.cache = EmbeddingCache()

    def embed_chunks(self, chunks: List[CodeChunk]):

        embeddings = []
        uncached_chunks = []
        uncached_indices = []

        for i, chunk in enumerate(chunks):
            cached = self.cache.get(chunk.chunk_id)

            if cached is not None:
                embeddings.append({
                    "chunk": chunk,
                    "embedding": cached
                })
            else:
                embeddings.append(None)
                uncached_chunks.append(chunk)
                uncached_indices.append(i)

        if uncached_chunks:
            new_embeddings = self.embedder.embed_chunks(uncached_chunks)

            for idx, chunk, emb in zip(
                uncached_indices,
                uncached_chunks,
                new_embeddings
            ):
                self.cache.set(chunk.chunk_id, emb)

                embeddings[idx] = {
                    "chunk": chunk,
                    "embedding": emb
                }

        return embeddings
```
